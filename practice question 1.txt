class marks
{

    {

        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
       
        
        if(a % 3 == 0)
        {
            System.out.println("Divisible by 3");
        }
        else if(a % 5 == 0)
        {
            System.out.println("Divisible by 5");
        
        }

        }
    }
}
